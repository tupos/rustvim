cmake_minimum_required(VERSION 3.16)
project(parser C)

add_compile_options(-w)

file(GLOB source_files src/*.c)

set(PARSER_LIBRARY_TYPE MODULE)
if(EMSCRIPTEN)
  set(PARSER_LIBRARY_TYPE STATIC)
endif()

add_library(parser
  ${PARSER_LIBRARY_TYPE}
  ${source_files}
)
set_target_properties(
  parser
  PROPERTIES
    OUTPUT_NAME ${PARSERLANG}
    PREFIX ""
)

include_directories(src)

install(TARGETS parser
  LIBRARY DESTINATION lib/nvim/parser
  ARCHIVE DESTINATION lib/nvim/parser)
